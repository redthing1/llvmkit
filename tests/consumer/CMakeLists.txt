cmake_minimum_required(VERSION 3.21)
project(llvmkit_consumer LANGUAGES CXX)

find_package(LLVM CONFIG REQUIRED)
add_executable(llvm_consumer llvm_consumer.cpp)
target_compile_features(llvm_consumer PRIVATE cxx_std_17)
target_include_directories(llvm_consumer PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(llvm_definitions NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(llvm_consumer PRIVATE ${llvm_definitions})
target_link_libraries(llvm_consumer PRIVATE LLVM)

if(LLVMKIT_WITH_LAB)
  find_package(remill CONFIG REQUIRED)
  find_package(LIEF CONFIG REQUIRED COMPONENTS SHARED)
  add_executable(remill_consumer remill_consumer.cpp)
  target_link_libraries(remill_consumer PRIVATE remill)
  add_executable(lief_consumer lief_consumer.cpp)
  target_link_libraries(lief_consumer PRIVATE LIEF::LIEF)
endif()
