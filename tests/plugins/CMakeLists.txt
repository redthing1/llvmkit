cmake_minimum_required(VERSION 3.21)
project(llvmkit_pass_plugins LANGUAGES C CXX)

find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_ENABLE_PLUGINS OR NOT LLVM_LINK_LLVM_DYLIB)
  message(FATAL_ERROR "This SDK must support plugins linked to shared LLVM")
endif()
list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
include(AddLLVM)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
set(CMAKE_CXX_STANDARD 17)
set(LLVM_REQUIRES_RTTI ${LLVM_ENABLE_RTTI})
set(LLVM_OPTIONAL_SOURCES input.c)
add_llvm_pass_plugin(LLVMKitPasses passes.cpp)
if(WIN32)
  set_target_properties(LLVMKitPasses PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
